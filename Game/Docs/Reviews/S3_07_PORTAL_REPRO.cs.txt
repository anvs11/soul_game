using System;
using System.Collections;
using Game.Actors;
using Game.Combat;
using Game.Core;
using Game.Sandbox;
using Game.Street;
using Game.UI;
using NUnit.Framework;
using UnityEngine;
using UnityEngine.SceneManagement;
using UnityEngine.TestTools;
using Object = UnityEngine.Object;
namespace Game.Tests
{
    public sealed class S307PortalReviewTests
    {
        private GameBootstrap boot;
        private GameObject blocker;
        private bool previousQueries;
        [UnityTest] public IEnumerator FailedPortalPlacementIsNotRetriedAfterBlockerRemoval()
        {
            previousQueries = Physics2D.queriesStartInColliders;
            Physics2D.queriesStartInColliders = true;
            foreach (var old in Object.FindObjectsByType<GameBootstrap>(FindObjectsSortMode.None)) Object.Destroy(old.gameObject);
            Time.timeScale = 1; yield return null;
            yield return SceneManager.LoadSceneAsync(ScenePaths.Bootstrap);
            yield return Until(() => (boot = Object.FindFirstObjectByType<GameBootstrap>()) != null && boot.Coordinator.Phase == RunPhase.Home);
            var home = Object.FindFirstObjectByType<HomeScreen>(); home.SeedInput.text = "0"; home.StreetButton.onClick.Invoke();
            yield return Until(() => boot.Coordinator.Phase == RunPhase.InArena);
            var arena = Object.FindFirstObjectByType<ArenaController>();
            var portal = Object.FindFirstObjectByType<StreetExitPortal>();
            var body = arena.Player.GetComponent<Rigidbody2D>(); body.simulated = false;
            blocker = new GameObject("S307 temporary World obstruction"); blocker.layer = LayerMask.NameToLayer("World");
            blocker.transform.position = arena.Player.transform.position;
            blocker.AddComponent<BoxCollider2D>().size = new Vector2(4, 4); Physics2D.SyncTransforms();
            foreach (var target in arena.Combatants)
                ((MonoBehaviour)target).GetComponent<DamageReceiver>().ApplyDamage(new DamagePacket(-1, 1000));
            Assert.That(arena.RemainingEnemies, Is.Zero); Assert.That(portal.IsOpen, Is.False);
            Object.Destroy(blocker); yield return null; Physics2D.SyncTransforms(); body.simulated = true;
            arena.TryExit(); yield return null;
            Assert.That(portal.IsOpen, Is.False, "Observed defect: no retry after temporary obstruction disappears.");
            Assert.That(boot.Coordinator.Phase, Is.EqualTo(RunPhase.InArena));
            Assert.That(portal.TryOpen(arena.Player), Is.True, "Placement now succeeds if explicitly retried.");
            TestContext.WriteLine("S307_REPRO: clear arena remains closed after obstruction removal; explicit TryOpen succeeds.");
            arena.Leave(); yield return Until(() => boot.Coordinator.Phase == RunPhase.Home);
        }
        [UnityTearDown] public IEnumerator Cleanup()
        {
            Physics2D.queriesStartInColliders = previousQueries; Time.timeScale = 1;
            if (blocker != null) Object.Destroy(blocker);
            foreach (var old in Object.FindObjectsByType<GameBootstrap>(FindObjectsSortMode.None)) Object.Destroy(old.gameObject);
            yield return null;
            var clean = SceneManager.CreateScene("S307ReviewCleanup"); SceneManager.SetActiveScene(clean);
            foreach (var name in new[] { ScenePaths.Home, ScenePaths.Stage3StreetSandbox })
            { var scene = SceneManager.GetSceneByName(name); if (scene.IsValid() && scene.isLoaded) yield return SceneManager.UnloadSceneAsync(scene); }
        }
        private static IEnumerator Until(Func<bool> predicate)
        { float limit = Time.realtimeSinceStartup + 15; while (!predicate()) { Assert.Less(Time.realtimeSinceStartup, limit); yield return null; } }
    }
}
